
import LawyersCard from "@/components/LawyersCard";
import {
    Search,
    SlidersHorizontal,
    BriefcaseBusiness,
    ArrowDownUp,
} from "lucide-react";

async function getLawyers() {
    const res = await fetch(`http://localhost:5000/lawyers`, {
        cache: "no-store",
    });

    if (!res.ok) {
        throw new Error("Failed to fetch lawyers");
    }

    return res.json();
}

export default async function LawyersPage() {
    const lawyers = await getLawyers();

    return (
        <section className="max-w-7xl mx-auto px-5 py-16">
            <div className="mb-16">

                {/* Heading */}

                <div className="text-center">

                    <h1 className="text-5xl font-bold text-slate-900">
                        Browse{" "}
                        <span className="text-amber-500">
                            Lawyers
                        </span>
                    </h1>

                    <p className="mt-4 max-w-2xl mx-auto text-lg text-slate-500">
                        Find the right legal expert for your needs.
                        Search, filter, and sort lawyers to quickly
                        connect with the best professional for your case.
                    </p>

                </div>

                {/* Search + Filters */}

                <div className="mt-12 bg-white rounded-3xl border border-slate-200 shadow-sm p-3">

                    <div className="grid lg:grid-cols-[2fr_1.2fr_1fr_1fr]">

                        {/* Search */}

                        <div className="flex items-center gap-3 px-5 border-r border-slate-200">

                            <Search
                                size={22}
                                className="text-slate-400"
                            />

                            <input
                                type="text"
                                placeholder="Search lawyers..."
                                className="w-full py-4 outline-none bg-transparent text-slate-700 placeholder:text-slate-400"
                            />

                        </div>

                        {/* Category */}

                        <div className="flex items-center gap-3 px-5 border-r border-slate-200">

                            <BriefcaseBusiness
                                size={22}
                                className="text-amber-500"
                            />

                            <select className="w-full bg-transparent py-4 outline-none cursor-pointer text-slate-700">

                                <option>All Categories</option>
                                <option>Corporate Law</option>
                                <option>Criminal Law</option>
                                <option>Family Law</option>
                                <option>Immigration Law</option>
                                <option>Business Law</option>

                            </select>

                        </div>

                        {/* Availability */}

                        <div className="flex items-center gap-3 px-5 border-r border-slate-200">

                            <SlidersHorizontal
                                size={22}
                                className="text-amber-500"
                            />

                            <select className="w-full bg-transparent py-4 outline-none cursor-pointer text-slate-700">

                                <option>Status</option>
                                <option>Available</option>
                                <option>Busy</option>

                            </select>

                        </div>

                        {/* Sort */}

                        <div className="flex items-center gap-3 px-5">

                            <ArrowDownUp
                                size={22}
                                className="text-amber-500"
                            />

                            <select className="w-full bg-transparent py-4 outline-none cursor-pointer text-slate-700">

                                <option>Sort By</option>
                                <option>Highest Rating</option>
                                <option>Most Experienced</option>
                                <option>Lowest Fee</option>
                                <option>Highest Fee</option>
                                <option>Newest Joined</option>

                            </select>

                        </div>

                    </div>

                </div>

                {/* Results */}

                <p className="text-center text-slate-500 mt-8">
                    Showing{" "}
                    <span className="font-semibold text-amber-500">
                        {lawyers.length}
                    </span>{" "}
                    lawyers
                </p>

            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-10">
                {lawyers.map((lawyer) => (
                    <LawyersCard key={lawyer._id} lawyer={lawyer} />
                ))}
            </div>
        </section>
    );
}
